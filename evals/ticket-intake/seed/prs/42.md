# PR #42 — feat(checkout): apply percentage coupon codes

**State:** open · **Author:** dev.marta · **Base:** main · **Ticket:** SHOP-101

## Description

Adds coupon input on `/checkout`. Valid percentage coupons reduce the subtotal. Unknown
and expired codes show inline errors. Codes are normalised to upper case before lookup.

- `src/checkout/coupon.ts`

+export type CouponResult =
+  | { ok: true; discount: number }
+  | { ok: false; error: 'Coupon not recognised' | 'Coupon expired' };
+
+const MAX_PERCENT = 50;
+
+export function applyCoupon(code: string, subtotal: number, now = new Date()): CouponResult {
+  const normalised = code.trim().toUpperCase();
+  const coupon = COUPONS[normalised];
+  if (!coupon) return { ok: false, error: 'Coupon not recognised' };
+  if (coupon.expiresAt < now) return { ok: false, error: 'Coupon expired' };
+  const percent = Math.min(coupon.percent, MAX_PERCENT);
+  return { ok: true, discount: Math.round(subtotal * percent) / 100 };
+}
+
+const COUPONS: Record<string, { percent: number; expiresAt: Date }> = {
+  SAVE10: { percent: 10, expiresAt: new Date('2027-01-01') },
+  SAVE60: { percent: 60, expiresAt: new Date('2027-01-01') },
+  OLD5: { percent: 5, expiresAt: new Date('2025-01-01') },
+};
